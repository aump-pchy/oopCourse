เงื่อนไขคืออะไร

เงื่อนไขคือนิพจน์ที่ได้ผลเป็น true หรือ false เท่านั้น สร้างจากตัวดำเนินการเปรียบเทียบและตรรกะที่เรียนในหน่วย 2

เงื่อนไขถามว่า
a <= 0a เป็นศูนย์หรือติดลบไหม
n % 2 == 1n เป็นเลขคี่ไหม (เศษจากหาร 2 เท่ากับ 1)
score >= 50 && score <= 100คะแนนอยู่ในช่วง 50 ถึง 100 ไหม
name.equals("Aump")ข้อความตรงกับ Aump ไหม

ขั้น 1 คำสั่ง if

        
สร้างคลาส SampleIf ใช้ JOptionPane ทำกล่องรับค่าและกล่องข้อความแบบง่าย

        
package th.ac.loeitech.unit3;

import javax.swing.JOptionPane;

public class SampleIf {
    public static void main(String[] args) {
        String input = JOptionPane.showInputDialog("กรอกคะแนน");
        int score = Integer.parseInt(input);

        if (score >= 50) {
            JOptionPane.showMessageDialog(null, "ผ่าน");
        }
        JOptionPane.showMessageDialog(null, "จบโปรแกรม");
    }
}

        
รัน 3 รอบ ป้อน 49, 50, 51 แล้วจดว่ารอบไหนเห็นคำว่า "ผ่าน" (null หมายถึงกล่องข้อความไม่มีหน้าต่างแม่)

ตัดเกรดจากคะแนน เรียงเงื่อนไขจากมากไปน้อย ถ้าสลับลำดับ ทุกคนจะได้ D หมด ลองดูสิ

        
String grade;
if (score >= 80) {
    grade = "A";
} else if (score >= 70) {
    grade = "B";
} else if (score >= 60) {
    grade = "C";
} else if (score >= 50) {
    grade = "D";
} else {
    grade = "F";
}
JOptionPane.showMessageDialog(null, "เกรด " + grade);

      


      

        
ขั้น 4 if ซ้อน (nested if)

        
if ซ้อนคือ if อยู่ข้างใน if อีกตัว ใช้เมื่อต้องผ่านด่านแรกก่อนถึงจะตัดสินด่านต่อไป ต่างจาก else-if ที่เป็นทางเลือกเรียงกัน

        
if (score >= 0 && score <= 100) {
    if (score >= 50) {
        JOptionPane.showMessageDialog(null, "ผ่าน");
    } else {
        JOptionPane.showMessageDialog(null, "ไม่ผ่าน");
    }
} else {
    JOptionPane.showMessageDialog(null, "คะแนนต้องอยู่ระหว่าง 0–100");
}

ขั้น 5 switch แบบมี break

        
ใช้เมื่อเทียบตัวแปรตัวเดียวกับค่าที่แน่นอนหลายค่า ใช้ได้กับ int, char, String และ enum

        
int level = Integer.parseInt(JOptionPane.showInputDialog("เลือกระดับ 1–3"));
switch (level) {
    case 1:
        JOptionPane.showMessageDialog(null, "ง่าย");
        break;
    case 2:
        JOptionPane.showMessageDialog(null, "ปานกลาง");
        break;
    case 3:
        JOptionPane.showMessageDialog(null, "ยาก");
        break;

}

        

          
ลองเอา break ออก

ขั้น 6 switch แบบลูกศร (Java 14+)

        
แบบใหม่ใช้ -> ไม่ต้องมี break และไม่ตกทะลุ ใช้เป็นนิพจน์ที่คืนค่าได้เลย รวมหลายค่าใน case เดียวด้วยจุลภาค

        
String level = JOptionPane.showInputDialog("เลือกระดับ: easy, normal, hard");
int maxNumber = switch (level) {
    case "easy" -> 10;
    case "normal" -> 50;
    case "hard" -> 100;
    default -> 10;
};
JOptionPane.showMessageDialog(null, "สุ่มเลข 1–" + maxNumber);

        
ข้อสอบและโค้ดเก่ายังใช้แบบมี break เยอะ ต้องอ่านเป็นทั้งสองแบบ แต่ถ้าเขียนใหม่เอง แนะนำแบบลูกศร

        คำใบ้ 1: สุ่มเลข 1–10 ยังไง
        
(int) (Math.random() * 10) ได้ 0–9 ต้อง + 1 ถึงจะได้ 1–10 หรือใช้ new java.util.Random().nextInt(1, 11) ซึ่งตัวเลขหลังไม่รวมในผลสุ่ม

if (ยังมีสิทธิ์ทาย) {

        // ลดสิทธิ์ แจ้งว่ามากกว่า
    } else if (ทายน้อยกว่าคำตอบ) {
        // ลดสิทธิ์ แจ้งว่าน้อยกว่า
    } else {
        // ถูกต้อง จบเกม
    }
    // ถ้าสิทธิ์เหลือ 0 และยังไม่ถูก → เฉลย จบเกม

ภารกิจหาบั๊ก

      
เพื่อนเขียนเกมแล้วส่งมาให้ช่วยดู โค้ดนี้คอมไพล์ผ่าน แต่มีบั๊กเชิงตรรกะ 2 จุด หาให้เจอก่อนเปิดเฉลย

      
private int answer = (int) (Math.random() * 10);
private int triesLeft = 3;

private void jButton1ActionPerformed(java.awt.event.ActionEvent evt) {
    int guess = Integer.parseInt(jTextField1.getText());
    if (triesLeft <= 0) {
        if (guess > answer) {
            triesLeft--;
            JOptionPane.showMessageDialog(this, "มากกว่าคำตอบ เหลือ " + triesLeft + " ครั้ง");
        } else if (guess < answer) {
            triesLeft--;
            JOptionPane.showMessageDialog(this, "น้อยกว่าคำตอบ เหลือ " + triesLeft + " ครั้ง");
        } else {
            JOptionPane.showMessageDialog(this, "ถูกต้อง!");
        }
    } else {
        JOptionPane.showMessageDialog(this, "หมดสิทธิ์ทายแล้ว คำตอบคือ " + answer);
    }
}

      

        เฉลยบั๊ก
        
บั๊ก 1: สุ่มได้ 0–9 ไม่ใช่ 1–10 เลข 10 ไม่มีวันเป็นคำตอบ และอาจได้ 0 แก้เป็น (int) (Math.random() * 10) + 1

        
บั๊ก 2: เงื่อนไขกลับด้าน triesLeft <= 0 ต้องเป็น triesLeft > 0 ไม่งั้นกดครั้งแรกก็ขึ้นหมดสิทธิ์ทันที

        
แถม: ผิดครั้งที่ 3 ยังไม่เฉลย ต้องกดอีกรอบถึงจะเห็น และทายถูกแล้วยังกดต่อได้
