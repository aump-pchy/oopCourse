for (int n = 50; n <= 100; n++) {
    if (n % 7 == 0) {
        System.out.println("เลขแรกที่หาร 7 ลงตัว: " + n);

// continue: ข้ามรอบนี้ ไปรอบถัดไป

int sum = 0;
int max = Integer.MIN_VALUE;
for (int k = 1; k <= 5; k++) {
    int n = Integer.parseInt(JOptionPane.showInputDialog("ตัวเลขตัวที่ " + k));
    sum += n;
    if (n > max) {
        max = n;
    }
}
double avg = (double) sum / 5;
JOptionPane.showMessageDialog(null, "ผลรวม " + sum + " เฉลี่ย " + avg + " มากสุด " + max);

        
Integer.MIN_VALUE คือค่าน้อยที่สุดของ int ตัวเลขแรกที่กรอกจึงมากกว่าแน่นอน ถ้าเริ่ม max = 0 แล้วกรอกแต่เลขติดลบ จะได้คำตอบผิด

กับดักที่เจอบ่อย

      

        ลืม i++ ใน while
        
i ไม่เคยเปลี่ยน เงื่อนไขจริงตลอดไป เกิด loop ไม่รู้จบ โปรแกรมค้าง ใน NetBeans กดปุ่มสี่เหลี่ยมแดงข้างหน้าต่าง Output หรือเมนู Run > Stop Build/Run

      

      

        for (int i = 1; i <= 10; i++); { print(i); }
        
; หลังวงเล็บทำให้ loop วนตัวเปล่า 10 รอบ แล้วบล็อกข้างหลังทำงานแค่ครั้งเดียว ในกรณีนี้คอมไพล์ไม่ผ่านด้วย เพราะ i หมดขอบเขตไปแล้ว

      

      

        ประกาศ int sum = 0; ไว้ข้างใน loop
        
sum ถูกสร้างใหม่และกลับเป็น 0 ทุกรอบ และใช้นอก loop ไม่ได้ ตัวสะสมต้องประกาศก่อนเข้า loop เสมอ (ทวนเรื่องขอบเขตตัวแปรหน่วย 2)

      

      

        for (int i = 0; i != 10; i += 3)
        
i เป็น 0, 3, 6, 9, 12, ... ไม่มีวันเท่ากับ 10 พอดี loop จึงไม่จบ ใช้ < หรือ <= ปลอดภัยกว่า !=

Lab 2 ใส่ PIN ได้ 3 ครั้ง

      
งานที่เหมาะกับ do-while เพราะต้องถามอย่างน้อย 1 ครั้งเสมอ สร้างคลาส PinCheck ที่มี main

      
final String CORRECT_PIN = "1234";
final int MAX_TRIES = 3;
int tries = 0;
boolean unlocked = false;

do {
    String pin = JOptionPane.showInputDialog("ใส่ PIN (เหลือ " + (MAX_TRIES - tries) + " ครั้ง)");
    tries++;
    if (CORRECT_PIN.equals(pin)) {
        unlocked = true;
    } else {
        JOptionPane.showMessageDialog(null, "PIN ไม่ถูกต้อง");
    }
} while (!unlocked && tries < MAX_TRIES);

if (unlocked) {
    JOptionPane.showMessageDialog(null, "ยินดีต้อนรับ");
} else {
    JOptionPane.showMessageDialog(null, "ผิดครบ " + MAX_TRIES + " ครั้ง บัตรถูกล็อก");
}

      
    
        
  • ทดสอบ 3 กรณี: ถูกครั้งแรก, ถูกครั้งที่ 3, ผิดครบ 3 ครั้ง
    • 
        
  • ตั้ง breakpoint ที่บรรทัด tries++ แล้วดูค่า tries กับ unlocked ในแต่ละรอบ
    • 
      

      

        คำถามชวนคิด: ทำไมใช้ CORRECT_PIN.equals(pin) ไม่ใช้ pin.equals(CORRECT_PIN)
        
ถ้ากด Cancel pin จะเป็น null แล้ว pin.equals(...) จะพัง แต่ "1234".equals(null) แค่ได้ false เอาค่าคงที่ไว้หน้าจึงปลอดภัยกว่า

คำใบ้

      

        คำใบ้ 1: ตรวจเลขคู่/คี่ยังไง
        
เลขคู่ใช้ n % 2 == 0 ส่วนเลขคี่ให้ใช้ n % 2 != 0 อย่าใช้ n % 2 == 1 เพราะใน Java -3 % 2 ได้ -1 เลขคี่ติดลบจะหายไปหมด (ทวนห้องทดลองการหารในหน่วย 2)

      

      

        คำใบ้ 2: โครงของปุ่มค้นหา
        
if (ช่องใดว่าง) { แจ้งเตือน; return; }
int lower = ..., upper = ...;
if (upper < lower) { แจ้งเตือน; return; }
if (ไม่ได้เลือกทั้งคู่และคี่) { แจ้งเตือน; return; }
txtOutput.setText("");
for (int n = lower; n <= upper; n++) {
    if (เลือกคู่ และ n เป็นเลขคู่ หรือ เลือกคี่ และ n เป็นเลขคี่) {
        txtOutput.append(n + " ");
    }
}

งานเสริม 2 สรุปคะแนนทั้งห้อง: รับคะแนนด้วย JOptionPane วนด้วย while จนกว่ากด Cancel แล้วแสดงจำนวนคน ค่าเฉลี่ย สูงสุด ต่ำสุด และจำนวนคนที่ได้ 50 ขึ้นไป อย่าลืมกรณีไม่มีใครกรอกเลย

      

    


    

      
ภารกิจหาบั๊ก

      
โปรแกรมนี้ตั้งใจรับคะแนน 5 คนแล้วหาค่าเฉลี่ย คอมไพล์ผ่าน แต่มีบั๊ก 2 จุด

      
int sum = 0;
int count = 0;
int i = 1;
while (i <= 5) {
    int score = Integer.parseInt(JOptionPane.showInputDialog("คะแนนคนที่ " + i));
    sum += score;
    count++;
}
double avg = sum / count;
JOptionPane.showMessageDialog(null, "ค่าเฉลี่ย " + avg);

      

        เฉลยบั๊ก
        
บั๊ก 1: ไม่มี i++ ใน loop โปรแกรมจะถาม "คะแนนคนที่ 1" ไปเรื่อย ๆ ไม่จบ

        
บั๊ก 2: sum / count เป็น int หาร int เศษทศนิยมหายก่อนเก็บลง double แก้เป็น (double) sum / count
